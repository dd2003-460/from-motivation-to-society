	\chapter{贡献分解：条件概率差}
	\label{ch:contribution}

	\begin{motivation}
		上一章遗留的问题：单链的概率会算了，但现实里 A 和 B 同时作用——各占多少？这是原书 part3 ch4 里那个「不够精确」问题的正式答案所在。\\
		\textbf{核心动机}：给出「贡献」的操作性定义 $C_A = P(D \mid A) - P(D \mid \neg A)$，处理多因一果的交互项，并诚实标注这个方法的全部边界。
	\end{motivation}

	\section{从「有/无」到定量}
	\label{sec:contrib-def}

	\textcolor{gray}{\textit{【骨架 · 追问链（终版）】$C_A = P(D \mid A) - P(D \mid \neg A)$。先讲清这个式子在做什么：反事实对比——「有 A 的世界」对「没有 A 的世界」。为什么这个差值有意义？→ 因为它是可干预的量：改变 A 的存在与否，这个差就是行为的杠杆。}}

	\textcolor{gray}{（正文待写：反事实推演是预测的最小单元（引言的承诺在此兑现）；「如果当初没有 X」= 计算一个 $C_X$。）}

		\textcolor{gray}{（待写：你有没有想过，为什么法庭问的是「如果他没有开枪，死者还会死吗」，而不是「他是不是坏人」？）}
	\end{readingnote}

	\section{可干预性 = 因果的定义}
	\label{sec:contrib-intervene}

	\textcolor{gray}{\textit{【骨架 · 追问链（终版）】$C_A \neq 0 \iff$ A 对 D 有因果作用。这是本书对「因果」的操作性定义。观察性数据能算吗？→ 只能估，不能保证——混杂因子会把 $C_A$ 污染，诚实列出。}}

	\textcolor{gray}{（正文待写：干预 do(A) 与观察 P(A|D) 的区别（辛普森悖论一例）；「我们能说什么/不能说什么」分列。）}

	\begin{questionbox}
		\textcolor{gray}{（待写：反驳位——「历史不能做实验，这套工具对历史无用？」回答：不能干预，但可以比较——自然实验与断点案例是把观察逼近干预的手段；第13章 §13.3 演示。）}
	\end{questionbox}

	\paragraph*{逻辑衔接}
	\textcolor{gray}{（待写）单因会算了。但 A 和 B 同时在场上，且互相纠缠——下一节「多因一果的分配」处理全书最难的诚实问题。}

	\section{多因一果的分配}
	\label{sec:contrib-interaction}

	\textcolor{gray}{\textit{【骨架 · 追问链（终版）】A、B 同时作用且有交互项。诚实处理：$C_A + C_B \neq$ 总效应，交互项必须显式。交互项归谁？→ 两种方案：Shapley 值（对称分摊）或边际贡献排序（保序不保量）；说明选择理由与各自代价。}}

	\section{归一化与可读性}
